cmake_minimum_required(VERSION 3.16)
project(vecmath VERSION 1.8.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(vecmath SHARED src/vecmath.cpp)
target_include_directories(vecmath PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
set_target_properties(vecmath PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 1)
install(TARGETS vecmath)
install(DIRECTORY include/ DESTINATION include)
